"""Same witness bytes on exact main and integration; no imports of newly added modules."""
from datetime import UTC, date, datetime, timedelta
from types import SimpleNamespace
from unittest.mock import AsyncMock
import api.main
import pytest
from api.routers import irrigation_recommendation as route
from api.canonical_water_state import CanonicalWaterState
from test_field_irrigation_recommendation_endpoint import _FakeConn, _patch

@pytest.mark.asyncio
async def test_explicit_manual_intent_reaches_immutable_candidate(monkeypatch):
    now = datetime.now(UTC)
    tenant = '11111111-1111-1111-1111-111111111111'
    user = SimpleNamespace(tenant_id=tenant,user_id='22222222-2222-2222-2222-222222222222')
    conn = _FakeConn(depletion_mm=60,decision='auto:daily_balance')
    fetchrow = conn.fetchrow
    async def fetch(sql,*args):
        row = await fetchrow(sql,*args)
        if row and 'FROM water_ledger' in sql: row['ledger_date'] = date.today()
        return row
    conn.fetchrow=fetch
    conn.execute=AsyncMock()
    _patch(monkeypatch,conn)
    monkeypatch.setattr(route,'has_permission',lambda *_: True,raising=False)
    canonical = CanonicalWaterState(schema_version='1',tenant_id=tenant,field_id='fld_1',season_id='season_2026',crop='wheat',growth_stage='mid',depletion_mm=60,depletion_confidence=.9,ledger_date=date.today().isoformat(),ledger_age_hours=6,taw_mm=100,raw_fraction=.5,raw_mm=50,root_depth_m=1,soil_texture='governed',forecast=[],evidence={'location':{'elevation_m':123.0,'lat':17.0,'lon':45.0}},quality_status='verified',operational_eligible=True,limitations=[],water_state_digest='a'*64,weather_snapshot_digest='b'*64,soil_profile_digest='c'*64,season_state_digest='d'*64)
    import api.canonical_water_state as water
    monkeypatch.setattr(water,'resolve_canonical_water_state',AsyncMock(return_value=canonical))
    monkeypatch.setattr(route,'_field_weather_snapshot',AsyncMock(return_value=dict(t_min_c=18,t_max_c=34,source='weather-engine-forecast',valid_time=date.today().isoformat(),day_of_year=date.today().timetuple().tm_yday,rain_recent_mm=0,forecast_rain_mm=0)))
    submit=AsyncMock(return_value={'authoritative':True,'persisted':True,'stage':'candidate','decision_id':'dec-test'})
    monkeypatch.setattr(route,'_submit_candidate_to_decision',submit)
    req=route.FieldIrrigationRequest(submit_to_decision=True,manual_execution=dict(season_id='season_2026',system_id='pivot-1',irrigated_area_ha=4,application_efficiency=.8,source_reference='Farm specification 17',observed_at=(now-timedelta(days=1)).isoformat(),valid_from=now.isoformat(),valid_until=(now+timedelta(hours=1)).isoformat()))
    await route.field_irrigation_recommendation('fld_1',req,user)
    candidate=submit.await_args.args[0]['decision_value']
    assert 'manual_execution_source' in candidate, 'live producer discarded explicit manual source'
    source=candidate['manual_execution_source']
    assert source['target_depth_mm']==pytest.approx(candidate['recommendation']['target_refill_mm']/.8)
    assert source['target_volume_m3']==pytest.approx(source['target_depth_mm']*40)
    assert source['water_truth']['canonical']['operational_eligible'] is True
