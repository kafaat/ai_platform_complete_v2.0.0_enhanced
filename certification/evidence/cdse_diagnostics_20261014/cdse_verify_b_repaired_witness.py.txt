"""Read-only shared-worktree review reproductions; temporary storage only."""
from __future__ import annotations
import asyncio
import fcntl
import importlib.util
import json
import os
from pathlib import Path
import sys
import tempfile

ROOT = Path('/workspace/scratch/7e02e5ed5cb5/cdse-worktree')
OUT = ROOT.parent / 'analysis'
sys.path[:0] = [str(ROOT / 'services/raster-service'), str(ROOT)]
import cdse_diagnostics as d
import cdse_singleflight
import numpy as np
import pytest
import raster_settings
import rasterio
from rasterio.transform import from_origin
import tile_render

spec = importlib.util.spec_from_file_location('cdse_review_helpers', ROOT / 'services/raster-service/test_cdse_durable_diagnostics.py')
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)

def receipt():
    return d.failure_receipt(tenant='review', field_id='field', index='ndvi', date_from='2026-10-14', date_to='2026-10-14', bbox=[0,0,2,2], geometry=None, max_cloud_pct=40, outcome='empty_raster')

findings = []
with tempfile.TemporaryDirectory(prefix='cdse-review-', dir=OUT) as temporary:
    root = Path(temporary)
    with pytest.MonkeyPatch.context() as mp:
        mp.setattr(raster_settings, 'UPLOAD_DIR', str(root / 'lock'))
        actual_open, actual_close, actual_unlink = os.open, os.close, Path.unlink
        lock_fds, closed = [], []
        def tracked_open(path, flags, *args):
            fd = actual_open(path, flags, *args)
            if str(path).endswith('write.lock'):
                lock_fds.append(fd)
            return fd
        def tracked_close(fd):
            closed.append(fd)
            return actual_close(fd)
        def failed_replace(*args):
            raise OSError('injected replace failure')
        def failed_cleanup(path, *args, **kwargs):
            if str(path).endswith('.tmp'):
                raise PermissionError('injected cleanup failure')
            return actual_unlink(path, *args, **kwargs)
        mp.setattr(d.os, 'open', tracked_open)
        mp.setattr(d.os, 'close', tracked_close)
        mp.setattr(d.os, 'replace', failed_replace)
        mp.setattr(Path, 'unlink', failed_cleanup)
        try:
            d.persist(receipt())
        except OSError as exc:
            error = type(exc).__name__
        fd2 = actual_open(root / 'lock/cdse_diagnostics/write.lock', os.O_RDWR)
        try:
            try:
                fcntl.flock(fd2, fcntl.LOCK_EX | fcntl.LOCK_NB)
                held = False
            except BlockingIOError:
                held = True
        finally:
            actual_close(fd2)
        leaked = lock_fds[0] not in closed
        if leaked:
            actual_close(lock_fds[0])
        findings.append({'id':'cleanup_leaks_lock', 'observed':held and leaked, 'exception':error, 'lock_retained_after_failure':held, 'lock_fd_not_closed':leaked})
    with pytest.MonkeyPatch.context() as mp:
        mp.setattr(raster_settings, 'UPLOAD_DIR', str(root / 'ring'))
        mp.setattr(d, 'MAX_RECEIPTS', 2)
        d.persist(receipt()); d.persist(receipt())
        directory = root / 'ring/cdse_diagnostics'
        before = {p.name:p.read_bytes() for p in directory.glob('*.json')}
        mp.setattr(d.os, 'replace', failed_replace)
        try:
            d.persist(receipt())
        except OSError:
            pass
        after = {p.name:p.read_bytes() for p in directory.glob('*.json')}
        findings.append({'id':'rotate_before_commit_loses_old_receipt', 'observed':len(after)<len(before), 'before':len(before), 'after':len(after), 'lost_receipts':sorted(set(before)-set(after))})
    path = root / 'one-pixel.tif'
    with rasterio.open(path,'w',driver='GTiff',width=1,height=1,count=1,dtype='float32',crs='EPSG:4326',transform=from_origin(0,2,1,1),nodata=float('nan')) as dst:
        dst.write(np.array([[0.2]],dtype='float32'),1)
    observation = tile_render.raster_content_observation(str(path))
    findings.append({'id':'degenerate_shape_fabricates_zero_count','observed':observation.get('observable_pixels')==0,'real_finite_pixels':1,'observation':observation})
    with pytest.MonkeyPatch.context() as mp:
        mp.setattr(raster_settings,'UPLOAD_DIR',str(root/'classification'))
        cdse_singleflight.cdse_tile_cache.clear()
        def failed_check(*args):
            raise RuntimeError('injected content measurement failure')
        mp.setattr(tile_render,'raster_has_observable_content',failed_check)
        assert asyncio.run(t._run(mp,payload=t._tiff([[0.2]*2]*2))) is None
        record=json.loads(next((root/'classification/cdse_diagnostics').glob('*.json')).read_text())
        findings.append({'id':'content_check_failure_named_empty','observed':record['outcome']=='empty_raster','outcome':record['outcome'],'after_mask':record['response']['after_mask']})

report={'findings':findings,'live_provider_contacted':False}
(OUT/'cdse_verify_b_repaired_witness.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(report,ensure_ascii=False))
